// ============= COMPONENTS =============

function MetricCard({ label, value, change, isPositive }: HeadlineMetric) {
  return (
    <Card>
      <VStack gap={2}>
        <Text type="label" color="secondary">
          {label}
        </Text>
        <Heading level={2}>{value}</Heading>
        <HStack gap={1} vAlign="center">
          <Icon
            icon={isPositive ? 'arrowUp' : 'arrowDown'}
            color={isPositive ? 'success' : 'error'}
            size="sm"
          />
          <Text type="supporting" color="secondary">
            {change} vs. last week
          </Text>
        </HStack>
      </VStack>
    </Card>
  );
}

function TrendLegendItem({
  series,
  label,
}: Readonly<{ series: 'thisWeek' | 'lastWeek'; label: string }>) {
  return (
    <HStack gap={2} vAlign="center">
      <Icon icon="stop" size="xsm" xstyle={trendDotStyles[series]} />
      <Text type="supporting" color="secondary">
        {label}
      </Text>
    </HStack>
  );
}

const trendSeriesLabels: Record<string, 'thisWeek' | 'lastWeek'> = {
  'This week': 'thisWeek',
  'Last week': 'lastWeek',
};

function WeeklyTrendTooltip({
  active,
  payload,
  label,
}: Readonly<{
  active?: boolean;
  payload?: Array<{ name: string; value: number }>;
  label?: string;
}>) {
  if (active !== true || payload === undefined || payload.length === 0) {
    return null;
  }
  return (
    <Card padding={3}>
      <VStack gap={1}>
        <Text type="supporting" color="secondary">
          {label}
        </Text>
        {payload.map((entry) => (
          <HStack key={entry.name} gap={2} vAlign="center">
            <Icon icon="stop" size="xsm" xstyle={trendDotStyles[trendSeriesLabels[entry.name]]} />
            <Text type="supporting">
              {entry.name}: ${entry.value}k
            </Text>
          </HStack>
        ))}
      </VStack>
    </Card>
  );
}

const weeklyTrendTooltip = <WeeklyTrendTooltip />;

function WeeklyTrendChart() {
  return (
    <VStack gap={3}>
      <ResponsiveContainer width="100%" height={280}>
        <LineChart data={weeklyTrend} margin={{ top: 5, right: 10, left: 0, bottom: 5 }}>
          <CartesianGrid
            horizontal
            vertical={false}
            stroke="var(--color-border, rgba(5, 54, 89, 0.1))"
          />
          <XAxis
            dataKey="day"
            tick={{ fontSize: 12, fill: 'var(--color-text-secondary, #4E606F)' }}
            axisLine={false}
            tickLine={false}
          />
          <YAxis
            tick={{ fontSize: 12, fill: 'var(--color-text-secondary, #4E606F)' }}
            axisLine={false}
            tickLine={false}
            width={40}
            tickFormatter={(value: number) => `$${value}k`}
          />
          <Tooltip
            content={weeklyTrendTooltip}
            cursor={{ stroke: 'var(--color-border, rgba(5, 54, 89, 0.1))' }}
          />
          <Line
            type="monotone"
            dataKey="thisWeek"
            name="This week"
            stroke={trendColors.thisWeek}
            strokeWidth={2}
            dot={{ r: 3 }}
          />
          <Line
            type="monotone"
            dataKey="lastWeek"
            name="Last week"
            stroke={trendColors.lastWeek}
            strokeWidth={2}
            strokeDasharray="4 4"
            dot={false}
          />
        </LineChart>
      </ResponsiveContainer>
      <HStack gap={6} vAlign="center">
        <TrendLegendItem series="thisWeek" label="This week" />
        <TrendLegendItem series="lastWeek" label="Last week" />
      </HStack>
    </VStack>
  );
}
